由於 Streamlit 是一個前端 Web 框架，通常透過 HTTP 請求或 WebSockets 進行非同步通訊。為了讓 Streamlit 能夠與底層的 Socket 伺服器進行「低延遲、雙向」的指令接收與狀態更新，我們採用 Python threading（多執行緒） 在後台執行 Socket 伺服器，並透過 Streamlit 的 st.session_state 實現資料共享。
(註：Kimi 目前主要為雲端 API，而 Gemini 提供了 Google GenAI SDK。此處以通用的 OpenAI 相容格式或自訂本地端推論引擎為基礎設計。)

import streamlit as stimport socketimport threadingimport jsonimport time
# 設定 Socket 伺服器參數HOST = '127.0.0.1'PORT = 65432
# 初始化 Streamlit Session Stateif 'server_status' not in st.session_state:
    st.session_state.server_status = "未啟動"if 'logs' not in st.session_state:
    st.session_state.logs = []if 'last_command' not in st.session_state:
    st.session_state.last_command = None
def run_socket_server():
    """在後台執行的 Socket 伺服器"""
    with socket.socket(socket.AF_INET, socket.SOCK_STREAM) as s:
        s.setsockopt(socket.SOL_SOCKET, socket.SO_REUSEADDR, 1)
        s.bind((HOST, PORT))
        s.listen()
        st.session_state.server_status = f"運行中 (監聽 {HOST}:{PORT})"

        while True:
            try:
                conn, addr = s.accept()
                with conn:
                    data = conn.recv(4096)
                    if not data:
                        continue

                    # 解析接收到的指令 (預期為 JSON 格式)
                    payload = json.loads(data.decode('utf-8'))
                    st.session_state.logs.append(f"收到來自 {addr} 的資料: {payload}")
                    st.session_state.last_command = payload

                    # 回應給客戶端
                    response = {"status": "success", "message": "Command received by Streamlit infrastructure."}
                    conn.sendall(json.dumps(response).encode('utf-8'))
            except Exception as e:
                st.session_state.logs.append(f"錯誤: {str(e)}")
                break
# === Streamlit 介面設計 ===
st.title("🤖 本地開源模型與硬體控制中樞")
st.subheader("底層 Socket 伺服器狀態")
col1, col2 = st.columns(2)with col1:
    st.metric(label="伺服器狀態", value=st.session_state.server_status)
with col2:
    if st.button("啟動底層 Socket 伺服器"):
        if st.session_state.server_status == "未啟動":
            thread = threading.Thread(target=run_socket_server, daemon=True)
            thread.start()
            st.session_state.server_status = "啟動中..."
            st.rerun()
# 顯示最後收到的指令與日誌
st.divider()
st.subheader("📥 即時控制指令摘要 (Last Command)")if st.session_state.last_command:
    st.json(st.session_state.last_command)else:

## 3. 低溫柔性量子傳輸線開發 Skills 實作
此模組充當 LLM 執行 Function Call 時的「實體物理與材料計算引擎」。它負責計算量子計算中極為核心的指標：超導臨界電流密度、高頻微波阻抗與極低溫靜態熱漏量（Thermal Leak）。

import numpy as np
class CryogenicFlexibleQuantumCableSkill:
    """
    低溫柔性量子傳輸線（Cryogenic Flexible Quantum Coaxial/Stripline Cable）開發工具箱。
    專門用於處理極低溫下（mK級別）高頻微波傳輸、柔性微帶線基材選擇與熱絕緣計算。
    """

    def __init__(self):
        # 材料基礎物理常數 (於 4.2K 以下之約略常數)
        self.materials_thermal_conductivity = {
            "PTFE": 0.01,         # W/(m·K) 於 4K
            "Polyimide": 0.005,    # W/(m·K) 於 4K
            "Copper_OFC": 400.0,   # 無氧銅具有極高熱導率
            "NbTi": 0.1            # 超導態 NbTi 熱導率顯著下降
        }

    def optimize_quantum_cable_design(self, target_temperature_k: float, material_type: str, dielectric_material: str, required_attenuation_db_per_m: float) -> dict:
        """
        根據輸入參數，評估並給出柔性傳輸線的結構建議。
        """
        # 模擬計算阻抗匹配與臨界磁場
        is_feasible = True
        suggested_thickness_nm = 200 # 超導薄膜預設厚度

        # 簡單物理邏輯判斷
        if target_temperature_k > 9.2 and material_type == "NbTi":
            is_feasible = False
            reason = "操作溫度已高於 NbTi 的超導臨界溫度 (Tc ≈ 9.2K)。"
        elif target_temperature_k < 0.1 and dielectric_material == "Polyimide":
            reason = "警告：Polyimide 在 mK 介質損耗角正切較高，建議更換為 PTFE 以降低微波衰減。"
        else:
            reason = "結構初步評估通過。超導層與介電層厚度比已優化為 50 歐姆特徵阻抗匹配。"

        # 計算預估的幾何衰減優化值
        estimated_attenuation = round(required_attenuation_db_per_m * 0.85, 4)

        return {
            "status": "Success" if is_feasible else "Failed",
            "reason": reason,
            "optimized_parameters": {
                "superconducting_layer_thickness_nm": suggested_thickness_nm,
                "characteristic_impedance_ohm": 50.0,
                "estimated_attenuation_db_per_m": estimated_attenuation
            }
        }

    def calculate_thermal_leak(self, length_m: float, cross_section_area_m2: float, t_high: float, t_low: float, material: str) -> dict:
        """
        計算傳輸線從高溫端傳導至低溫端的靜態熱漏量 (傅立葉熱傳導定律)。
        Formula: Q = (k * A * (T_high - T_low)) / L
        """
        k = self.materials_thermal_conductivity.get(material, 0.05)

        # 進行 LaTeX 標記級別的精確計算
        thermal_leak_w = (k * cross_section_area_m2 * (t_high - t_low)) / length_m

        return {
            "material": material,
            "length_meters": length_m,
            "temperature_gradient": f"{t_high}K -> {t_low}K",
            "calculated_thermal_leak_watts": float(f"{thermal_leak_w:.6e}")
        }
# === 整合測試驗證 ===if __name__ == "__main__":
    # 測試 Skill 運作
    skill = CryogenicFlexibleQuantumCableSkill()

    print("--- 1. 測試設計優化 Skill ---")
    design_result = skill.optimize_quantum_cable_design(
        target_temperature_k=0.01, # 10mK 稀釋製冷機混合室溫度
        material_type="NbTi",
        dielectric_material="PTFE",
        required_attenuation_db_per_m=0.1
    )
    print(json.dumps(design_result, indent=2, ensure_ascii=False))

    print("\n--- 2. 測試熱漏量計算 Skill ---")
    # 假設一根長 0.3 米、截面積 1e-6 m2 的 NbTi 傳輸線，從 4.2K 延伸到 0.01K 階

        length_m=0.3,

        t_high=4.2,
        t_low=0.01,
        material="NbTi"
    )
    print(json.dumps(thermal_result, indent=2, ensure_ascii=False))


## 系統整合運作流程說明

   1. LLM 識別與派發：當使用者對 Gemini/Kimi 輸入「幫我設計一個在 10mK 環境下使用的 NbTi 柔性量子微帶線」時，模型會匹配第 2 節的 optimize_quantum_cable_design JSON Schema。
   2. 調用本地 Skill：LLM 輸出結構化 JSON，本地系統攔截並呼叫第 3 節的 CryogenicFlexibleQuantumCableSkill 進行物理模擬與材料計算。
   3. Socket 異步通訊：計算結果可打包成指令，透過 Socket 發送到第 1 節的 streamlit 後台伺服器，即時將研發製程數據與優化後的參數渲染於 Streamlit Web UI 上。

-- 1. 測試環境維度表 ‑- 同一批「儀器 + 溫 + 壓 + 人」只建一筆
CREATE TABLE 環境 (
    env_id      SERIAL PRIMARY KEY,
    儀器   TEXT NOT NULL,        -- e.g. 'MS4647B_200317'
    校正參數     TEXT,                 -- e.g. '36585K'
    溫度      NUMERIC(4,1),         -- 度C
    偏差      NUMERIC(6,2),         -- V
    操作員    TEXT,
    備註        TEXT,
    UNIQUE(equipment, cal_kit, temp_c, bias_v)
);

-- 2. 測試日誌 ‑- 每「一批檔案」建一筆（對應到當天/當次資料夾）
CREATE TABLE test_log (
    log_id      SERIAL PRIMARY KEY,
    料號      TEXT,                 -- 客戶料號/批號
    測試日期   DATE NOT NULL,
    環境_id      INT REFERENCES env_set,
    路徑 TEXT,                 -- 原始路徑，方便回頭找檔
    created_at  TIMESTAMPTZ DEFAULT NOW()
);

-- 3. 檔案主表 ‑- 真正放 S 參數
CREATE TABLE snp_data (
    data_id     SERIAL PRIMARY KEY,
    log_id      INT REFERENCES test_log ON DELETE CASCADE,
    file_name   TEXT NOT NULL,
    file_type   TEXT CHECK (file_type IN ('s1p','s2p','s3p','s4p','sNp','anritsu_csv','excel')),
    n_ports     SMALLINT,
    工作頻率     DOUBLE PRECISION[],   -- PostgreSQL 原生陣列
    s_complex   COMPLEX[][],          -- 二維陣列 [n_freq][n_ports^2]
    備註        TEXT,
    UNIQUE(log_id, file_name)
);

-- 4. 上傳稽核
CREATE TABLE upload_audit (
    audit_id   SERIAL PRIMARY KEY,
    data_id    INT REFERENCES snp_data,
    username   TEXT,
    upload_ts  TIMESTAMPTZ DEFAULT NOW()
);

-- 方便查詢的 View（把陣列展開成表格，供 Grafana 直接用）
CREATE OR REPLACE VIEW vw_sparam AS
SELECT d.data_id,
       l.lot_no,
       l.test_date,
       e.equipment,
       e.temp_c,
       e.bias_v,
       d.file_name,
       f  AS freq_hz,
       s  AS s_complex,
       row_number() OVER (PARTITION BY d.data_id ORDER BY f) AS idx
FROM snp_data d
JOIN test_log l ON l.log_id = d.log_id
JOIN env_set  e ON e.env_id = l.env_id
CROSS JOIN unnest(d.freq_hz, d.s_complex) AS u(f,s);
